== production/mkosi.extra/usr/lib/systemd/system/tundravm-root-mode.service
[Unit]
Description=Make the initramfs root root:root 0755
Before=sysinit.target
DefaultDependencies=no

[Service]
Type=oneshot
RemainAfterExit=yes
ExecStartPre=/usr/bin/chown root:root /
ExecStart=/usr/bin/chmod 0755 /
TimeoutStartSec=30s
NoNewPrivileges=yes
ProtectKernelModules=yes
ProtectKernelTunables=yes
ProtectControlGroups=yes
MemoryDenyWriteExecute=yes
ProtectKernelLogs=yes
ProtectClock=yes
ProtectHostname=yes
RestrictRealtime=yes
RestrictNamespaces=yes
LockPersonality=yes
RestrictAddressFamilies=AF_UNIX
SystemCallArchitectures=native
CapabilityBoundingSet=CAP_CHOWN CAP_FOWNER
PrivateNetwork=yes

[Install]
WantedBy=minimal.target
== production/mkosi.postinst.d/50-postinst.sh
#!/usr/bin/env bash
set -euo pipefail

mkosi-chroot systemctl enable tundravm-root-mode.service
mkdir -p "$BUILDROOT/etc/systemd/system/minimal.target.wants"
ln -sf "/etc/systemd/system/tundravm-root-mode.service" "$BUILDROOT/etc/systemd/system/minimal.target.wants/"

== staging/mkosi.extra/etc/nftables.conf
#!/usr/sbin/nft -f
# tundravm Firewall: inbound traffic is dropped unless a rule below accepts it;
# outbound traffic is not filtered.
# The cloud metadata servers take only what the units that declare them send.
flush ruleset

table inet filter {
	chain input {
		type filter hook input priority filter; policy drop;
		ct state established,related accept
		ct state invalid drop
		iifname "lo" accept
		icmpv6 type { destination-unreachable, packet-too-big, time-exceeded, parameter-problem, nd-router-advert, nd-neighbor-solicit, nd-neighbor-advert } accept
		udp sport 67 udp dport 68 accept comment "DHCPv4 replies"
		ip6 saddr fe80::/10 udp sport 547 udp dport 546 accept comment "DHCPv6 replies"
		tcp dport 22 accept comment "ssh.service"
		tcp dport 8080 accept comment "api.service"
		tcp dport 8745 accept comment "attested-endpoint"
		tcp dport 9090 accept comment "api.service"
	}
	chain forward {
		type filter hook forward priority filter; policy drop;
	}
	chain output {
		type filter hook output priority filter; policy accept;
		ip daddr 169.254.169.254 udp dport 123 accept comment "metadata time"
		ip daddr { 168.63.129.16, 169.254.169.254 } drop comment "cloud metadata"
		ip6 daddr fd20:ce::254 drop comment "cloud metadata"
	}
}
