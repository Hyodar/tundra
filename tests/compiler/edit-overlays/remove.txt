#!/usr/bin/env bash
set -euo pipefail

mkosi-chroot useradd --system --shell /usr/sbin/nologin api
mkosi-chroot useradd --system --shell /usr/sbin/nologin tick
mkosi-chroot useradd --system --shell /usr/sbin/nologin web

== staging/mkosi.extra/etc/nftables.conf
#!/usr/sbin/nft -f
# tundravm Firewall: inbound traffic is dropped unless a rule below accepts it;
# outbound traffic is not filtered.
# The cloud metadata servers take only what the units that declare them send.
flush ruleset

table inet filter {
	chain input {
		type filter hook input priority filter; policy drop;
		ct state established,related accept
		ct state invalid drop
		iifname "lo" accept
		icmpv6 type { destination-unreachable, packet-too-big, time-exceeded, parameter-problem, nd-router-advert, nd-neighbor-solicit, nd-neighbor-advert } accept
		udp sport 67 udp dport 68 accept comment "DHCPv4 replies"
		ip6 saddr fe80::/10 udp sport 547 udp dport 546 accept comment "DHCPv6 replies"
		tcp dport 22 accept comment "ssh.service"
		tcp dport 8745 accept comment "attested-endpoint"
	}
	chain forward {
		type filter hook forward priority filter; policy drop;
	}
	chain output {
		type filter hook output priority filter; policy accept;
		ip daddr 169.254.169.254 udp dport 123 accept comment "metadata time"
		ip daddr { 168.63.129.16, 169.254.169.254 } drop comment "cloud metadata"
		ip6 daddr fd20:ce::254 drop comment "cloud metadata"
	}
}
