== production/mkosi.extra/usr/lib/systemd/system/db-ready.service
[Unit]
Description=Wait until db.service is ready
After=db.service
Requires=db.service

[Service]
Type=oneshot
RemainAfterExit=yes
ExecStart=/bin/bash -c "until (exec 3<>/dev/tcp/127.0.0.1/5432) 2>/dev/null; do sleep 0.5; done"
User=db-ready
TimeoutStartSec=90s
ProtectSystem=strict
ProtectHome=yes
PrivateTmp=yes
NoNewPrivileges=yes
ProtectKernelModules=yes
ProtectKernelTunables=yes
ProtectControlGroups=yes
RestrictSUIDSGID=yes
MemoryDenyWriteExecute=yes
PrivateDevices=yes
ProtectKernelLogs=yes
ProtectClock=yes
ProtectHostname=yes
RestrictRealtime=yes
RestrictNamespaces=yes
LockPersonality=yes
RestrictAddressFamilies=AF_UNIX AF_INET AF_INET6
SystemCallArchitectures=native
SystemCallFilter=@system-service
SystemCallFilter=~@privileged @resources
SystemCallFilter=@setuid
SystemCallErrorNumber=EPERM
CapabilityBoundingSet=
UMask=0077

[Install]
WantedBy=minimal.target
== production/mkosi.extra/usr/lib/systemd/system/tundravm-root-mode.service
[Unit]
Description=Make the initramfs root root:root 0755
Before=sysinit.target
DefaultDependencies=no

[Service]
Type=oneshot
RemainAfterExit=yes
ExecStartPre=/usr/bin/chown root:root /
ExecStart=/usr/bin/chmod 0755 /
TimeoutStartSec=30s
NoNewPrivileges=yes
ProtectKernelModules=yes
ProtectKernelTunables=yes
ProtectControlGroups=yes
MemoryDenyWriteExecute=yes
ProtectKernelLogs=yes
ProtectClock=yes
ProtectHostname=yes
RestrictRealtime=yes
RestrictNamespaces=yes
LockPersonality=yes
RestrictAddressFamilies=AF_UNIX
SystemCallArchitectures=native
CapabilityBoundingSet=CAP_CHOWN CAP_FOWNER
PrivateNetwork=yes

[Install]
WantedBy=minimal.target
== production/mkosi.postinst.d/50-postinst.sh
#!/usr/bin/env bash
set -euo pipefail

mkosi-chroot useradd --system --shell /usr/sbin/nologin db-ready
mkosi-chroot systemctl enable db-ready.service
mkosi-chroot systemctl enable tundravm-root-mode.service
mkdir -p "$BUILDROOT/etc/systemd/system/minimal.target.wants"
ln -sf "/etc/systemd/system/db-ready.service" "$BUILDROOT/etc/systemd/system/minimal.target.wants/"
ln -sf "/etc/systemd/system/tundravm-root-mode.service" "$BUILDROOT/etc/systemd/system/minimal.target.wants/"
