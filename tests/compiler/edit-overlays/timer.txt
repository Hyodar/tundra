== production/mkosi.extra/usr/lib/systemd/system/tundravm-power-button.service

[Service]
Type=simple
ExecStart=/usr/sbin/acpid --foreground --netlink --nosocket --logevents --confdir /usr/lib/tundravm/acpi --pidfile /run/tundravm-power-button/acpid.pid
Restart=always
RestartSec=5
ProtectSystem=strict
ProtectHome=yes
PrivateTmp=yes
NoNewPrivileges=yes
ProtectKernelModules=yes
ProtectKernelTunables=yes
ProtectControlGroups=yes
MemoryDenyWriteExecute=yes
ProtectKernelLogs=yes
ProtectClock=yes
ProtectHostname=yes
RestrictRealtime=yes
RestrictNamespaces=yes
LockPersonality=yes
RestrictAddressFamilies=AF_NETLINK AF_UNIX
SystemCallArchitectures=native
CapabilityBoundingSet=
RuntimeDirectory=tundravm-power-button

[Install]
WantedBy=minimal.target
== production/mkosi.extra/usr/lib/systemd/system/tundravm-root-mode.service

== production/mkosi.extra/usr/lib/tundravm/acpi/power-button
event=button[ /]power
action=/usr/bin/systemctl poweroff
== production/mkosi.postinst.d/50-postinst.sh
#!/usr/bin/env bash
set -euo pipefail

mkosi-chroot systemctl enable tundravm-root-mode.service
mkosi-chroot systemctl enable tundravm-power-button.service
mkdir -p "$BUILDROOT/etc/systemd/system/minimal.target.wants"
ln -sf "/etc/systemd/system/tundravm-root-mode.service" "$BUILDROOT/etc/systemd/system/minimal.target.wants/"
ln -sf "/etc/systemd/system/tundravm-power-button.service" "$BUILDROOT/etc/systemd/system/minimal.target.wants/"
mkosi-chroot systemctl mask acpid.path acpid.service acpid.socket
